class UploadAPIClient:
    
    def consultar_lote(self, lote_id):
        """Consulta status de um lote"""
        url = f"{self.base_url}/api/lotes/{lote_id}/status"
        
        response = requests.get(
            url,
            headers={"Authorization": f"Bearer {self.api_key}"},
            timeout=30
        )
        
        if response.status_code == 200:
            return True, response.json()
        else:
            return False, {"message": f"Erro {response.status_code}"}
    
    def download_arquivo(self, download_url, save_path):
        """Baixa nota fiscal"""
        response = requests.get(
            download_url,
            headers={"Authorization": f"Bearer {self.api_key}"},
            timeout=60,
            stream=True
        )
        
        if response.status_code == 200:
            with open(save_path, 'wb') as f:
                for chunk in response.iter_content(chunk_size=8192):
                    f.write(chunk)
            return True, save_path
        else:
            return False, f"Erro {response.status_code}"

job_consultar_notas.py:

def processar_lotes():
    # Buscar lotes pendentes
    lotes = db.get_lotes_upload_pendente()
    
    for lote in lotes:
        # Consultar na API
        success, result = upload_api.consultar_lote(lote['id'])
        
        if success and result.get('status') == 'completed':
            # Baixar arquivo
            nf = result['nota_fiscal']
            save_path = f"uploads/nota_fiscal_lote_{lote['id']}.pdf"
            
            download_success, path = upload_api.download_arquivo(
                nf['download_url'],
                save_path
            )
            
            if download_success:
                # Atualizar banco
                db.atualizar_status_upload(lote['id'], 'completed')
                db.salvar_nota_fiscal(lote['id'], path)
                print(f"✅ N.F. do lote {lote['id']} recebida!")
